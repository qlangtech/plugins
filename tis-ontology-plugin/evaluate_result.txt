1.nl:qid:294:哪些商店在开业三年后的库存周转率高于或等于同类商店平均水平？------------------------------------------------
1.expect sql------------------------------------------------
WITH store_opening_dates AS (
    SELECT
        store_id,
        store_open_date AS open_date,
        date(store_open_date, '+36 months') AS third_year_date
    FROM
        toy_stores
),
valid_sales AS (
    SELECT
        s.store_id,
        SUM(CAST(s.units AS INTEGER)) AS total_units
    FROM
        toy_sales s
        JOIN store_opening_dates sod ON s.store_id = sod.store_id
    WHERE
        s.date >= sod.third_year_date
    GROUP BY
        s.store_id
),
store_inventory AS (
    SELECT
        store_id,
        SUM(CAST(stock_on_hand AS INTEGER)) AS total_stock
    FROM
        toy_inventory
    GROUP BY
        store_id
),
turnover_rates AS (
    SELECT
        vs.store_id,
        (vs.total_units * 1.0) / si.total_stock AS turnover_rate,
        st.store_city
    FROM
        valid_sales vs
        JOIN store_inventory si ON vs.store_id = si.store_id
        JOIN toy_stores st ON vs.store_id = st.store_id
),
city_avg_turnover AS (
    SELECT
        store_city,
        AVG(turnover_rate) AS avg_rate
    FROM
        turnover_rates
    GROUP BY
        store_city
)
SELECT
    tr.store_id,
    tr.turnover_rate AS "库存周转率",
    cat.avg_rate AS "同类平均周转率"
FROM
    turnover_rates tr
    JOIN city_avg_turnover cat ON tr.store_city = cat.store_city
    JOIN toy_stores st ON tr.store_id = st.store_id
WHERE
    tr.turnover_rate >= cat.avg_rate
1.generate------------------------------------------------
WITH store_turnover AS (
  SELECT
    st.Store_ID,
    st.Store_Name,
    st.Store_City,
    SUM(s.Units * CAST(REPLACE(TRIM(p.Product_Price), '$', '') AS DECIMAL(18,2))) AS total_sales_value,
    SUM(i.Stock_On_Hand * CAST(REPLACE(TRIM(p.Product_Price), '$', '') AS DECIMAL(18,2))) AS total_stock_value,
    COALESCE(SUM(s.Units * CAST(REPLACE(TRIM(p.Product_Price), '$', '') AS DECIMAL(18,2))) / NULLIF(SUM(i.Stock_On_Hand * CAST(REPLACE(TRIM(p.Product_Price), '$', '') AS DECIMAL(18,2))), 0), 0) AS turnover_rate
  FROM toy_stores AS st
  JOIN toy_inventory AS i ON st.Store_ID = i.Store_ID
  JOIN toy_products AS p ON i.Product_ID = p.Product_ID
  JOIN toy_sales AS s ON i.Store_ID = s.Store_ID AND i.Product_ID = s.Product_ID
  WHERE date_trunc('day', CAST(REPLACE(st.Store_Open_Date, '/', '-') AS DATE)) <= date_trunc('day', DATE_SUB(CURRENT_DATE(), INTERVAL 3 YEAR))
  GROUP BY st.Store_ID, st.Store_Name, st.Store_City
),
city_avg AS (
  SELECT
    Store_City,
    AVG(turnover_rate) AS avg_turnover_rate
  FROM store_turnover
  GROUP BY Store_City
)
SELECT
  st.Store_Name
FROM store_turnover AS st
JOIN city_avg AS ca ON st.Store_City = ca.Store_City
WHERE st.turnover_rate >= ca.avg_turnover_rate;

2.nl:qid:299:如何列出2007年第四季度开业的店铺中Art & Crafts类商品的销售总利润？------------------------------------------------
2.expect sql------------------------------------------------
WITH stores_opened_in_q4 AS (
    SELECT store_id
    FROM toy_stores
    WHERE Store_Open_Date >= '2007-10-01' 
      AND Store_Open_Date <= '2007-12-31'
),
art_products AS (
    SELECT 
        product_id,
        CAST(REPLACE(REPLACE(Product_Price, '$', ''), ' ', '') AS REAL) AS product_price,
        CAST(REPLACE(REPLACE(Product_Cost, '$', ''), ' ', '') AS REAL) AS product_cost
    FROM toy_products
    WHERE Product_Category = 'Art & Crafts'
)
SELECT
    SUM(CAST(s.Units AS INTEGER) * (ap.product_price - ap.product_cost)) AS `销售总利润`
FROM
    toy_sales s
    INNER JOIN stores_opened_in_q4 st ON s.Store_ID = st.store_id
    INNER JOIN art_products ap ON s.Product_ID = ap.product_id;
2.generate------------------------------------------------
SELECT 
  SUM(s.Units * (CAST(REPLACE(TRIM(p.Product_Price), '$', '') AS DECIMAL(18,2)) - CAST(p.Product_Cost AS DECIMAL(18,2)))) AS total_profit
FROM toy_sales s
JOIN toy_products p ON s.Product_ID = p.Product_ID
JOIN toy_stores st ON s.Store_ID = st.Store_ID
WHERE 
  p.Product_Category = 'Art & Crafts'
  AND YEAR(CAST(REPLACE(st.Store_Open_Date, '/', '-') AS DATE)) = 2007
  AND QUARTER(CAST(REPLACE(st.Store_Open_Date, '/', '-') AS DATE)) = 4

3.nl:qid:303:如何找出库存量高于平均水平的门店及其所在城市？------------------------------------------------
3.expect sql------------------------------------------------
with store_total_stock as (
    select
        store_id,
        sum(cast(stock_on_hand as int)) as total_stock
    from
        toy_inventory
    group by
        store_id
)
select
    s.store_id as `store_id`,
    s.store_city as `store_city`
from
    toy_stores s
inner join
    store_total_stock sts
on
    s.store_id = sts.store_id
where
    sts.total_stock > (
        select
            avg(total_stock)
        from
            store_total_stock
    )
3.generate------------------------------------------------
SELECT st.Store_id , st.Store_Name, st.Store_City
FROM toy_stores AS st
JOIN (
  SELECT i.Store_ID
  FROM toy_inventory AS i
  GROUP BY i.Store_ID
  HAVING SUM(i.Stock_On_Hand) > (SELECT AVG(Stock_On_Hand) FROM toy_inventory)
) AS filtered_stores ON st.Store_ID = filtered_stores.Store_ID;
